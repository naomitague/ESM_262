compute_diversity
function (species) 
{
    species = as.factor(species)
    tmp = (summary(species)/sum(summary(species)))^2
    diversity = sum(tmp)
    return(diversity)
}
possible.fish = as.factor(c("salmon","steelhead","shark","tuna","cod"))

Using lists to return multiple items from a function

We can return a list from a function to give the user multiple pieces of information

# a function that returns the max and min of a vector of numbers
minmax = function(x) {
  minx = min(x)
  maxx = max(x)
  return(list(min=minx, max=maxx))
}

minmax(c(3,5,2,8,1))
$min
[1] 1

$max
[1] 8

summarize_diversity
function (species) 
{
    species = as.factor(species)
    tmp = (summary(species)/sum(summary(species)))^2
    diversity = 1 - sum(tmp)
    nspecies = length(summary(species))
    tmp = which.max(summary(species))
    dominant = names(summary(species)[tmp])
    return(list(num = nspecies, simpson = diversity, dominant = dominant))
}
# run function and save results to a variable
d1 = summarize_diversity(catch1)
d2 = summarize_diversity(catch2)
